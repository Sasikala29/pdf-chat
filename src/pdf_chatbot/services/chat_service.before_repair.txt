
"""Answer PDF questions using retrieval-augmented generation."""

from src.pdf_chatbot.core.constants import (
    DEFAULT_MIN_SIMILARITY,
    DEFAULT_RETRIEVAL_TOP_K,
    NO_ANSWER,
)
from src.pdf_chatbot.repositories.sqlite_repository import (
    load_chat_history,
    load_embedded_chunks,
    save_chat_message,
)
from src.pdf_chatbot.services.embedding_service import embed_query
from src.pdf_chatbot.services.llm_service import generate_answer
from src.pdf_chatbot.services.retrieval_service import retrieve_relevant_chunks


def answer_question(
    document_id: str,
    question: str,
    conversation_id: str | None = None,
) -> str:
    """Answer a PDF question and persist it in the selected conversation.

    Args:
        document_id: ID of the indexed PDF used as the answer source.
        question: The user's question.
        conversation_id: Optional ID of the conversation receiving the messages.
            When omitted, the repository retains backward-compatible behavior.

    Returns:
        The generated answer, including source-page information when available.

    Raises:
        ValueError: If the question is blank or the document has no indexed text.
        RuntimeError: If embedding or answer generation fails.
    """
    cleaned_question = question.strip()
    if not cleaned_question:
        raise ValueError("Enter a question before submitting.")

    stored_chunks = load_embedded_chunks(document_id)
    if not stored_chunks:
        raise ValueError("No indexed text was found for this document.")

    if conversation_id is None:
        save_chat_message(document_id, "user", cleaned_question)
    else:
        save_chat_message(
            document_id,
            "user",
            cleaned_question,
            conversation_id=conversation_id,
        )

    query_embedding = embed_query(cleaned_question)
    retrieved_chunks = retrieve_relevant_chunks(
        query_embedding,
        stored_chunks,
        top_k=DEFAULT_RETRIEVAL_TOP_K,
        min_similarity=DEFAULT_MIN_SIMILARITY,
    )

    if not retrieved_chunks:
        answer = NO_ANSWER
        source_pages: list[int] = []
    else:
        context_chunks = [
            (item.item.chunk.page_number, item.item.chunk.text)
            for item in retrieved_chunks
        ]
        generated_answer = generate_answer(cleaned_question, context_chunks)
        answer = generated_answer.strip() or NO_ANSWER

        if answer == NO_ANSWER:
            source_pages = []
        else:
            # The current generator does not return structured citations, so
            # these are retrieved pages, not guaranteed sentence-level citations.
            source_pages = list(
                dict.fromkeys(
                    item.item.chunk.page_number for item in retrieved_chunks
                )
            )
            answer = (
                f"{answer}\n\nSource pages: "
                f"{', '.join(map(str, source_pages))}"
            )

    if conversation_id is None:
        save_chat_message(
            document_id,
            "assistant",
            answer,
            source_pages=source_pages,
        )
    else:
        save_chat_message(
            document_id,
            "assistant",
            answer,
            source_pages=source_pages,
            conversation_id=conversation_id,
        )

    return answer


def get_document_chat_history(
    document_id: str,
    conversation_id: str | None = None,
) -> list[dict[str, object]]:
    """Return chronological history for a PDF or a specific conversation."""
    return load_chat_history(document_id, conversation_id)
